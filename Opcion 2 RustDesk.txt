# ==================================================
# Instalacion de RustDesk via MSI - equipos administrados Windows 10/11
# Funciona ejecutado a mano (con elevacion) o como SYSTEM
# (Intune, GPO, PDQ, tarea programada, psexec -s)
# Log: C:\ProgramData\RustDeskDeploy\deploy.log
# ==================================================
#Requires -Version 5.1

$Version     = "1.4.6"
$MsiUrl      = "https://github.com/rustdesk/rustdesk/releases/download/$Version/rustdesk-$Version-x86_64.msi"
$InstallDir  = "C:\Program Files\RustDesk"
$RdExe       = Join-Path $InstallDir "rustdesk.exe"
$ServiceName = "RustDesk"
$Password    = "Sistemas..22"
$WorkDir     = Join-Path $env:ProgramData "RustDeskDeploy"
$MsiPath     = Join-Path $WorkDir "rustdesk.msi"
$MsiLog      = Join-Path $WorkDir "msi-install.log"

$Identity    = [Security.Principal.WindowsIdentity]::GetCurrent()
$IsSystem    = $Identity.IsSystem
$IsAdmin     = ([Security.Principal.WindowsPrincipal]$Identity).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
$Interactive = [Environment]::UserInteractive -and -not $IsSystem

function Exit-Script([int]$Code) {
    try { Stop-Transcript | Out-Null } catch {}
    if ($Interactive) { Read-Host "Presiona Enter para salir" | Out-Null }
    exit $Code
}

function Fail([string]$Msg) {
    Write-Host "ERROR: $Msg" -ForegroundColor Red
    Exit-Script 1
}

# --- Elevacion (solo si se ejecuta a mano sin admin) ---
if (-not $IsAdmin) {
    if (-not $PSCommandPath) {
        Write-Host "Guarda este codigo como .ps1 y ejecutalo desde el archivo." -ForegroundColor Red
        exit 1
    }
    try {
        Start-Process powershell.exe -Verb RunAs -ErrorAction Stop `
            -ArgumentList "-NoProfile -ExecutionPolicy Bypass -File `"$PSCommandPath`""
    } catch {
        Write-Host "No se pudo obtener permisos de administrador." -ForegroundColor Red
        Write-Host "Ejecutalo con una cuenta admin o despliegalo como SYSTEM (Intune/GPO/PDQ)."
        Read-Host "Presiona Enter para salir" | Out-Null
        exit 1
    }
    exit 0
}

New-Item -ItemType Directory -Path $WorkDir -Force | Out-Null
Start-Transcript -Path (Join-Path $WorkDir "deploy.log") -Append | Out-Null

Write-Host "=== Instalador RustDesk $Version (MSI) ===" -ForegroundColor Cyan
Write-Host "Equipo: $env:COMPUTERNAME | Usuario: $($Identity.Name) | Arquitectura: $env:PROCESSOR_ARCHITECTURE"

# --- 1. Limpieza de instalaciones previas ---
Write-Host "[1/6] Limpiando instalaciones previas..."
$svc = Get-Service -Name $ServiceName -ErrorAction SilentlyContinue
if ($svc -and $svc.Status -ne 'Stopped') {
    Stop-Service $ServiceName -Force -ErrorAction SilentlyContinue
    try { $svc.WaitForStatus('Stopped', [TimeSpan]::FromSeconds(20)) } catch {}
}
Get-Process -Name "rustdesk" -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
Start-Sleep -Seconds 2

# Desinstalar versiones MSI previas (si las hay)
$uninstallKeys = @(
    'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*',
    'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*'
)
$previous = Get-ItemProperty $uninstallKeys -ErrorAction SilentlyContinue |
    Where-Object { $_.DisplayName -like 'RustDesk*' -and $_.PSChildName -match '^\{[0-9A-Fa-f\-]+\}$' }
foreach ($p in $previous) {
    Write-Host "   Desinstalando MSI previo $($p.DisplayName) $($p.DisplayVersion)..."
    Start-Process msiexec.exe -ArgumentList "/x $($p.PSChildName) /qn /norestart" -Wait
}

# Restos de una instalacion con el .exe: quitar servicio y carpeta
if (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue) {
    sc.exe delete $ServiceName | Out-Null
    for ($i = 0; $i -lt 20 -and (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue); $i++) {
        Start-Sleep -Seconds 1
    }
    if (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue) {
        Fail "El servicio anterior quedo 'marcado para eliminacion'. Cierra services.msc o reinicia el equipo y vuelve a ejecutar."
    }
}
if (Test-Path $InstallDir) {
    Remove-Item $InstallDir -Recurse -Force -ErrorAction SilentlyContinue
}

# --- 2. Descarga (con reintentos) ---
Write-Host "[2/6] Descargando MSI..."
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$ProgressPreference = 'SilentlyContinue'
$downloaded = $false
for ($try = 1; $try -le 3 -and -not $downloaded; $try++) {
    try {
        Invoke-WebRequest -Uri $MsiUrl -OutFile $MsiPath -UseBasicParsing -ErrorAction Stop
        if ((Get-Item $MsiPath).Length -gt 1MB) { $downloaded = $true }
    } catch {
        Write-Host "   Intento $try fallido: $($_.Exception.Message)" -ForegroundColor Yellow
        Start-Sleep -Seconds 5
    }
}
if (-not $downloaded) { Fail "No se pudo descargar $MsiUrl (revisa proxy/firewall)." }
Unblock-File -Path $MsiPath -ErrorAction SilentlyContinue
Write-Host "   Descarga OK."

# --- 3. Instalacion MSI (espera real a que termine) ---
Write-Host "[3/6] Instalando..."
$msi = Start-Process msiexec.exe -Wait -PassThru `
    -ArgumentList "/i `"$MsiPath`" /qn /norestart /l*v `"$MsiLog`""
if ($msi.ExitCode -notin 0, 3010, 1641) {
    Fail "msiexec termino con codigo $($msi.ExitCode). Revisa $MsiLog"
}
if (-not (Test-Path $RdExe)) { Fail "No se encontro $RdExe despues de instalar." }
Write-Host "   Instalado (codigo $($msi.ExitCode))."

# --- 4. Servicio ---
Write-Host "[4/6] Verificando servicio..."
for ($i = 0; $i -lt 30 -and -not (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue); $i++) {
    Start-Sleep -Seconds 1
}
if (-not (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue)) {
    Write-Host "   El MSI no registro el servicio, instalandolo manualmente..."
    & $RdExe --install-service | Out-Null
    Start-Sleep -Seconds 5
}
if (-not (Get-Service -Name $ServiceName -ErrorAction SilentlyContinue)) {
    Fail "No existe el servicio $ServiceName."
}
Set-Service -Name $ServiceName -StartupType Automatic

$started = $false
for ($try = 1; $try -le 3 -and -not $started; $try++) {
    try {
        if ((Get-Service $ServiceName).Status -ne 'Running') {
            Start-Service $ServiceName -ErrorAction Stop
        }
        (Get-Service $ServiceName).WaitForStatus('Running', [TimeSpan]::FromSeconds(15))
        $started = $true
    } catch {
        Write-Host "   Intento $try de arranque fallido." -ForegroundColor Yellow
        Start-Sleep -Seconds 5
    }
}
if (-not $started) {
    Write-Host "   Eventos recientes del Service Control Manager:" -ForegroundColor Yellow
    Get-WinEvent -FilterHashtable @{ LogName = 'System'; ProviderName = 'Service Control Manager'; StartTime = (Get-Date).AddMinutes(-15) } -ErrorAction SilentlyContinue |
        Where-Object { $_.Message -match 'RustDesk' } |
        Select-Object -First 5 | ForEach-Object { Write-Host "   $($_.TimeCreated): $($_.Message)" }
    Fail "El servicio no arranco. Revisa antivirus/Defender (historial de proteccion)."
}
Write-Host "   Servicio: $((Get-Service $ServiceName).Status)"

# --- 5. Contrasena permanente ---
Write-Host "[5/6] Configurando contrasena..."
Start-Sleep -Seconds 3
& $RdExe --password $Password | Out-Null   # el pipe hace que PowerShell espere

# --- 6. Obtener ID ---
Write-Host "[6/6] Obteniendo ID..."
$rdId = $null
for ($i = 0; $i -lt 5 -and -not $rdId; $i++) {
    $rdId = (& $RdExe --get-id | Out-String).Trim()
    if (-not $rdId) { Start-Sleep -Seconds 2 }
}

Remove-Item $MsiPath -Force -ErrorAction SilentlyContinue

Write-Host "===============================" -ForegroundColor Green
Write-Host "  Equipo : $env:COMPUTERNAME"   -ForegroundColor Cyan
Write-Host "  ID     : $rdId"               -ForegroundColor Cyan
Write-Host "===============================" -ForegroundColor Green
Exit-Script 0
